import logging
from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional

logger = logging.getLogger(__name__)


@dataclass
class Config:
    input_dir: Path
    output_dir: Path
    max_size: int = 100 * 1024 * 1024
    formats: list[str] = field(default_factory=lambda: ["pdf", "docx", "png"])
    verbose: bool = False


def load_config(path: str) -> Config:
    """Read the configuration file and return a Config object."""
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)
    return Config(
        input_dir=Path(data["input_dir"]),
        output_dir=Path(data["output_dir"]),
        max_size=data.get("max_size", 100 * 1024 * 1024),
        formats=data.get("formats", ["pdf"]),
        verbose=data.get("verbose", False),
    )


class FileProcessor:
    def __init__(self, config: Config):
        self.config = config
        self.processed = 0
        self.errors: list[str] = []

    def find_files(self):
        for root, dirs, files in os.walk(self.config.input_dir):
            for name in files:
                ext = name.rsplit(".", 1)[-1].lower()
                if ext in self.config.formats:
                    yield Path(root) / name

    def process(self, path: Path) -> Optional[Path]:
        size = path.stat().st_size
        if size > self.config.max_size:
            self.errors.append(f"{path}: file is too large ({size} bytes)")
            return None
        target = self.config.output_dir / path.name
        try:
            target.write_bytes(path.read_bytes())
        except OSError as e:
            logger.error("failed to process %s: %s", path, e)
            self.errors.append(str(e))
            return None
        self.processed += 1
        return target

    def run(self) -> int:
        self.config.output_dir.mkdir(parents=True, exist_ok=True)
        for path in self.find_files():
            result = self.process(path)
            if result and self.config.verbose:
                print(f"done: {result}")
        print(f"{self.processed} files processed, {len(self.errors)} errors")
        return 0 if not self.errors else 1


if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("usage: process.py CONFIG", file=sys.stderr)
        sys.exit(2)
    sys.exit(FileProcessor(load_config(sys.argv[1])).run())

export function formatBytes(bytes, decimals = 1) {
  if (bytes === 0) return "0 B";
  const units = ["B", "KB", "MB", "GB", "TB"];
  const i = Math.floor(Math.log(bytes) / Math.log(1024));
  return `${(bytes / Math.pow(1024, i)).toFixed(decimals)} ${units[i]}`;
}

export function parseUser(input: unknown): Result<User> {
  if (typeof input !== "object" || input === null) {
    return { ok: false, error: "expected an object" };
  }
  const obj = input as Record<string, unknown>;
  if (typeof obj.id !== "number" || typeof obj.name !== "string") {
    return { ok: false, error: "missing id or name" };
  }
  return {
    ok: true,
    value: {
      id: obj.id,
      name: obj.name,
      email: String(obj.email ?? ""),
      createdAt: new Date(String(obj.createdAt)),
    },
  };
}

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class FileService {
    private final Path root;
    private final List<String> history = new ArrayList<>();

    public FileService(Path root) {
        this.root = root;
    }

    public long totalSize() throws IOException {
        try (var stream = Files.walk(root)) {
            return stream.filter(Files::isRegularFile)
                    .mapToLong(p -> {
                        try {
                            return Files.size(p);
                        } catch (IOException e) {
                            return 0L;
                        }
                    })
                    .sum();
        }
    }

    public void rename(String from, String to) throws IOException {
        Files.move(root.resolve(from), root.resolve(to));
        history.add(from + " -> " + to);
    }

    @Override
    public String toString() {
        return "FileService{root=" + root + ", changes=" + history.size() + "}";
    }
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define BUFFER_SIZE 4096

typedef struct {
    char *data;
    size_t len;
    size_t cap;
} buffer_t;

static int buffer_append(buffer_t *b, const char *src, size_t n) {
    if (b->len + n > b->cap) {
        size_t cap = b->cap ? b->cap * 2 : BUFFER_SIZE;
        while (cap < b->len + n) cap *= 2;
        char *p = realloc(b->data, cap);
        if (p == NULL) return -1;
        b->data = p;
        b->cap = cap;
    }
    memcpy(b->data + b->len, src, n);
    b->len += n;
    return 0;
}

int main(int argc, char **argv) {
    if (argc < 2) {
        fprintf(stderr, "usage: %s FILE\n", argv[0]);
        return 1;
    }
    FILE *f = fopen(argv[1], "rb");
    if (!f) {
        perror("fopen");
        return 1;
    }
    buffer_t buf = {0};
    char chunk[BUFFER_SIZE];
    size_t n;
    while ((n = fread(chunk, 1, sizeof(chunk), f)) > 0) {
        if (buffer_append(&buf, chunk, n) != 0) {
            fprintf(stderr, "out of memory\n");
            fclose(f);
            return 1;
        }
    }
    fclose(f);
    printf("%zu bytes read\n", buf.len);
    free(buf.data);
    return 0;
}

#[derive(Debug, Clone, PartialEq)]
pub struct Entry {
    pub name: String,
    pub size: u64,
    pub modified: i64,
}

pub fn count_words(path: &Path) -> io::Result<HashMap<String, usize>> {
    let file = File::open(path)?;
    let mut counts = HashMap::new();
    for line in BufReader::new(file).lines() {
        for word in line?.split_whitespace() {
            *counts.entry(word.to_lowercase()).or_insert(0) += 1;
        }
    }
    Ok(counts)
}

impl Entry {
    pub fn new(name: &str, size: u64) -> Self {
        Entry { name: name.to_string(), size, modified: 0 }
    }

    pub fn is_empty(&self) -> bool {
        self.size == 0
    }
}

자주 묻는 질문

파일은 어떻게 변환하나요?
사이트에서 필요한 도구를 고른 다음, 파일을 업로드 영역으로 끌어다 놓으면 됩니다. 버튼을 눌러 컴퓨터에서 파일을 직접 선택해도 됩니다. 변환이 끝나면 다운로드 버튼이 자동으로 나타납니다.

무료인가요?
네, 기본 도구는 모두 무료입니다. 회원 가입이 필요 없고 결과 파일에 워터마크도 붙지 않습니다.

제 파일은 안전한가요?
업로드한 파일은 안전하게 처리되며 한 시간이 지나면 자동으로 삭제됩니다. 파일 내용을 다른 곳에 공유하거나 판매하지 않습니다. 많은 도구가 브라우저 안에서 바로 작동하기 때문에 파일이 내 기기 밖으로 나가지 않습니다.

변환이 실패하는 이유는 무엇인가요?
파일이 손상되었거나, 비밀번호가 걸려 있거나, 지원하지 않는 형식인 경우가 대부분입니다. 파일을 확인한 뒤 다시 시도해 보세요. 그래도 문제가 계속되면 고객센터로 문의해 주세요.

한글 파일(HWP)도 열 수 있나요?
네. HWP와 HWPX 파일을 PDF나 워드 문서로 바꿀 수 있고, 브라우저에서 바로 내용을 확인할 수도 있습니다.

서울시, 자전거 전용도로 25km 늘린다

서울시는 앞으로 2년 동안 자전거 전용도로 25km를 새로 만들겠다고 지난 화요일 밝혔다. 총사업비는 약 160억 원으로, 도심과 주요 주거 지역, 대학가를 잇는 것이 목표다.

시 관계자는 "출퇴근이나 통학에 자전거를 이용하려는 시민이 꾸준히 늘고 있다"며 "안전하게 달릴 수 있는 길을 만드는 것이 가장 중요하다"고 말했다. 찬성하는 쪽에서는 교통 체증과 미세먼지를 줄이는 데 도움이 될 것이라고 기대한다. 반면 일부 상인들은 주차 공간이 줄어들까 걱정하고 있다.

공사는 내년 봄에 시작할 예정이며, 시는 동네마다 주민 설명회를 열어 세부 설계를 함께 논의할 계획이다.

재택근무를 할 때 집중력을 지키는 방법

재택근무는 출퇴근 시간을 아끼고 하루를 스스로 조절할 수 있다는 점에서 장점이 많습니다. 하지만 집에서는 쉽게 산만해지고, 일과 생활의 경계가 흐려지기도 합니다. 몇 가지 습관만 바꿔도 훨씬 편해집니다.

먼저 일하는 자리를 정해 두세요. 꼭 방이 따로 있을 필요는 없습니다. 조용한 구석에 책상 하나만 있어도 충분합니다. 그 자리에 앉으면 일하는 시간이라고 몸이 기억하게 됩니다.

다음으로 시작하는 시간과 끝내는 시간을 매일 같게 지켜 보세요. 중간중간 짧게 쉬는 것도 중요합니다. 5분 정도 걷거나 물 한 잔을 마시기만 해도 머리가 맑아집니다.

마지막으로 퇴근을 확실하게 하세요. 일이 끝나면 노트북을 덮고 전혀 다른 일을 해 보세요. 남은 일은 내일 해도 괜찮습니다.

보안 경고가 나타나면 공식 홈페이지에서 받은 파일인지 확인한 다음 실행을 누르세요. 프로그램을 지우려면 설정에서 앱 목록을 열고 해당 프로그램을 찾아 제거를 누르면 됩니다.

How to work from home without losing your focus

Working from home has become normal for millions of people. It saves time, reduces stress, and gives us more control over our day. But it also brings new problems: distractions, long hours, and the feeling that work never really ends. Here are some simple habits that can help.

First, create a dedicated workspace. It does not have to be a separate room. A small desk in a quiet corner is enough, as long as you use it only for work. When you sit there, your brain knows it is time to focus.

Second, keep a regular schedule. Start and finish at the same time every day, and take short breaks. A five-minute walk or a glass of water can make a big difference.

Third, plan your day the evening before. Write down the three most important tasks for tomorrow. In the morning, start with the hardest one, before checking your email or messages.

Finally, remember to log off. Close your laptop at the end of the day and do something completely different. Your work will still be there tomorrow.

Understanding variables and functions

A variable is a name that refers to a value stored in memory. When you write x = 10, you create a variable called x and give it the value 10. Later, you can use x anywhere in your program, and the computer will replace it with its current value. Variables can hold numbers, text, lists, and many other kinds of data.

A function is a reusable block of code that performs a specific task. You define it once and call it as many times as you need. Functions can take input values, called parameters or arguments, and they can return a result. For example, a function named add might take two numbers and return their sum.

Using functions makes your code easier to read, test, and maintain. Instead of repeating the same lines in several places, you write them once inside a function. If you later find a bug or want to change the behavior, you only need to update one place.

Simple banana bread

Ingredients: 3 ripe bananas, 1/3 cup melted butter, 3/4 cup sugar, 1 egg, 1 teaspoon vanilla extract, 1 teaspoon baking soda, a pinch of salt, 1 1/2 cups all-purpose flour.

Preheat the oven to 175 degrees Celsius (350 degrees Fahrenheit) and grease a loaf pan. In a large bowl, mash the bananas with a fork. Stir in the melted butter, then the sugar, the egg, and the vanilla. Sprinkle the baking soda and salt over the mixture and stir. Add the flour and mix until just combined. Pour the batter into the pan and bake for about 60 minutes, or until a toothpick inserted into the center comes out clean. Let it cool for 10 minutes before slicing.

Five tips for a better night's sleep

Most adults need between seven and nine hours of sleep each night, but many people get much less. Poor sleep affects your mood, your memory, and even your immune system. The good news is that small changes can make a big difference.

Go to bed and wake up at the same time every day, even on weekends. A regular rhythm helps your body know when it is time to rest. Avoid screens for at least an hour before bed, because the light from phones and computers makes it harder to fall asleep. Keep your bedroom dark, quiet, and cool. Limit caffeine after lunch, and try not to eat a heavy meal late in the evening. If you cannot fall asleep after twenty minutes, get up, read a book in dim light, and go back to bed when you feel sleepy.

How compound interest works

Compound interest means that you earn interest not only on the money you save, but also on the interest you have already earned. Imagine that you put 1,000 dollars into a savings account that pays 5 percent per year. After one year, you have 1,050 dollars. In the second year, you earn 5 percent on 1,050 dollars, not on 1,000, so you end the year with 1,102.50 dollars. Over a long time, this effect becomes very large. After 30 years, the same 1,000 dollars would grow to more than 4,300 dollars without any new deposits. That is why it is a good idea to start saving early, even if you can only save small amounts.

A short history of the printing press

Before the fifteenth century, books in Europe were copied by hand, usually by monks working in monasteries. A single book could take months to finish, and only rich people and institutions could afford them. Around 1440, Johannes Gutenberg, a goldsmith from Mainz, developed a printing press with movable metal type. Each letter was cast as a small piece of metal that could be arranged into lines and pages, inked, and pressed onto paper.

The new technology spread quickly. By 1500, printing presses were operating in more than 250 cities, and millions of books had been produced. Books became cheaper, more people learned to read, and new ideas could travel faster than ever before. Many historians believe that the printing press played an important role in the Renaissance, the Reformation, and the scientific revolution.
